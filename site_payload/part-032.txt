function arrangeClusterLayout(nodes,edges){
  const cx=W/2,cy=H/2,minDim=Math.min(W,H);
  const nodeMap=new Map(nodes.map(n=>[n.id,n]));
  const adj=new Map(nodes.map(n=>[n.id,new Set()]));
  const coreLinks=new Map(nodes.map(n=>[n.id,new Set()]));
  for(const e of edges){
    if(!adj.has(e.a)||!adj.has(e.b))continue;
    adj.get(e.a).add(e.b);adj.get(e.b).add(e.a);
    if(coreIds.has(e.a)&&!coreIds.has(e.b))coreLinks.get(e.b).add(e.a);
    if(coreIds.has(e.b)&&!coreIds.has(e.a))coreLinks.get(e.a).add(e.b);
  }
  const corePos={};
  if(focusSet.size===1){
    const fid=[...focusSet][0];corePos[fid]={x:cx,y:cy};const others=CORE.filter(([id])=>id!==fid);const r=Math.min(205,minDim*.27);const ang0=-Math.PI/2;others.forEach(([id],i)=>{const a=ang0+i*(Math.PI*2/others.length);corePos[id]=polar(cx,cy,r,a)})
  }else if(focusSet.size>1){
    const selected=CORE.filter(([id])=>focusSet.has(id)),unselected=CORE.filter(([id])=>!focusSet.has(id));const r=Math.min(150,minDim*.20);selected.forEach(([id],i)=>{const a=-Math.PI/2+i*(Math.PI*2/selected.length);corePos[id]=polar(cx,cy,r,a)});const r2=Math.min(255,minDim*.34);unselected.forEach(([id],i)=>{const a=-Math.PI/2+i*(Math.PI*2/Math.max(1,unselected.length));corePos[id]=polar(cx,cy,r2,a)})
  }else{
    const r=Math.min(255,minDim*.34);const ang0=-Math.PI/2;CORE.forEach(([id],i)=>{const a=ang0+i*(Math.PI*2/CORE.length);corePos[id]=polar(cx,cy,r,a)})
  }
  for(const [id,p] of Object.entries(corePos)){const n=nodeMap.get(id);if(n){n.x=p.x;n.y=p.y;n.fx=null;n.fy=null}}

  // Shared nodes (connected directly to 2+ of the five core bodies) form a compact inner ring.
  const shared=nodes.filter(n=>!coreIds.has(n.id)&&coreLinks.get(n.id)?.size>=2);
  shared.sort((a,b)=>{
    const d=(coreLinks.get(b.id)?.size||0)-(coreLinks.get(a.id)?.size||0);
    return d||a.name.localeCompare(b.name,'ru');
  });
  const ringR=Math.min(145,minDim*.20);
  shared.forEach((n,i)=>{
    const d=coreLinks.get(n.id).size;
    const ring=1+Math.floor(i/14);
    const rr=ringR+(ring-1)*55;
    const a=-Math.PI/2+(i%14)*(Math.PI*2/14);
    const jitter=((hashString(n.id)%1000)/1000-.5)*0.10;
    const p=polar(cx,cy,rr,a+jitter); n.x=p.x;n.y=p.y;n.fx=null;n.fy=null;
  });

  // Nodes connected to one core are placed in that core's sector.
  const byCore=new Map(CORE.map(([id])=>[id,[]]));
  const unanchored=[];
  for(const n of nodes){
    if(coreIds.has(n.id)||shared.includes(n))continue;
    const links=[...(coreLinks.get(n.id)||[])];
    if(links.length===1)byCore.get(links[0]).push(n);else unanchored.push(n);
  }
  for(const [coreId,list] of byCore){
    const cp=corePos[coreId]||{x:cx,y:cy};
    const base=Math.atan2(cp.y-cy,cp.x-cx);
    // Spread nodes on a broad fan away from the center.
    list.sort((a,b)=>hashString(a.id)-hashString(b.id));
    const maxPerRing=Math.max(8,Math.ceil(Math.sqrt(list.length+1)*2.2));
    list.forEach((n,i)=>{
      const ring=Math.floor(i/maxPerRing);
      const pos=i%maxPerRing;
      const count=Math.min(maxPerRing,list.length-ring*maxPerRing);
      const spread=Math.min(Math.PI*0.82, 1.15 + ring*0.10);
      const a=base-spread/2 + (count<=1?.5:(pos/(count-1)))*spread;
      const rr=Math.min(140+ring*78,minDim*.46);
      const p=polar(cp.x,cp.y,rr,a);
      n.x=p.x;n.y=p.y;n.fx=null;n.fy=null;
    })
  }

  // Remaining nodes form a stable outer ring, grouped deterministically.
  unanchored.sort((a,b)=>hashString(a.id)-hashString(b.id));
  const outerR=Math.min(335,minDim*.48);
  const nOuter=Math.max(1,unanchored.length);
  unanchored.forEach((n,i)=>{
    const a=-Math.PI/2+i*(Math.PI*2/nOuter);
    const rr=outerR+(i%3)*45;
    const p=polar(cx,cy,rr,a);
    n.x=p.x;n.y=p.y;n.fx=null;n.fy=null;
  });

  // В режиме фокуса одного узла выбранный объект остаётся в центре.
  // Основные органы, общие хабы и прочие внешние участники разводятся по компактным концентрическим уровням.
  if(focusSet.size===1){
    const fid=[...focusSet][0];
    const fp=nodeMap.get(fid);
    if(fp){
      fp.x=cx; fp.y=cy;
      const ringCore=[],ringShared=[],ringOuter=[];
      for(const n of nodes){
        if(n.id===fid) continue;
        if(coreIds.has(n.id)) ringCore.push(n);
        else if(sharedCoreDegree(n.id)>=2) ringShared.push(n);
        else ringOuter.push(n);
      }
      ringCore.sort((a,b)=>(a.name||'').localeCompare(b.name||'','ru'));
      ringShared.sort((a,b)=>sharedCoreDegree(b.id)-sharedCoreDegree(a.id)||(a.name||'').localeCompare(b.name||'','ru'));
      ringOuter.sort((a,b)=>String(a.type||'').localeCompare(String(b.type||''))||(a.name||'').localeCompare(b.name||'','ru'));
      const placeCompactRings=(list,r0,step,maxPerRing,maxR,startPhase=0)=>{
        if(!list.length)return;
        let offset=0,ring=0;
        while(offset<list.length){
          const count=Math.min(maxPerRing,list.length-offset);
          const rr=Math.min(r0+ring*step,maxR);
          const phase=startPhase+(ring%2?Math.PI/Math.max(8,count):0);
          for(let pos=0;pos<count;pos++){
            const n=list[offset+pos];
            const a=phase+(count===1?0:(pos/count)*Math.PI*2);
            const p=polar(fp.x,fp.y,rr,a);
            n.x=p.x; n.y=p.y; n.fx=null; n.fy=null;
          }
          offset+=count; ring++;
        }
      };
      const maxCore=Math.min(172,minDim*.31);
      const maxShared=Math.min(245,minDim*.44);
      const maxOuter=Math.min(330,minDim*.59);
      placeCompactRings(ringCore,148,54,10,maxCore,-Math.PI/2);
      placeCompactRings(ringShared,218,56,10,maxShared,-Math.PI/2+Math.PI/12);
      placeCompactRings(ringOuter,292,58,12,maxOuter,-Math.PI/2+Math.PI/24);
      resolveNodeOverlaps(nodes,new Set([fid]),42,26);
      return;
    }
  }
  if(focusSet.size>1){
    // Multi-focus: selected ministries form a compact, human-readable group near the center.
    // Each selected ministry gets its own full 360° halo; shared nodes sit in a separate overlap ring.
    const selectedIds=[...focusSet];
    const selectedSet=new Set(selectedIds);
    const selectedNodes=selectedIds.map(id=>nodeMap.get(id)).filter(Boolean);

    // Place selected ministries in a compact pattern rather than on a vertical line.
    const spreadBase=Math.min(230,minDim*.28);
    if(selectedNodes.length===2){
      selectedNodes[0].x=cx-spreadBase/2; selectedNodes[0].y=cy;
      selectedNodes[1].x=cx+spreadBase/2; selectedNodes[1].y=cy;
    }else if(selectedNodes.length===3){
      const rr=Math.min(160,minDim*.20);
      selectedNodes.forEach((n,i)=>{const a=-Math.PI/2+i*(Math.PI*2/3); const p=polar(cx,cy,rr,a); n.x=p.x; n.y=p.y;});
    }else{
      const rr=Math.min(175,minDim*.22);
      selectedNodes.forEach((n,i)=>{const a=-Math.PI/2+i*(Math.PI*2/selectedNodes.length); const p=polar(cx,cy,rr,a); n.x=p.x; n.y=p.y;});
    }

    const directByCore=new Map(selectedIds.map(id=>[id,[]]));
    const sharedSelected=[];
    const unselectedCoreNodes=[];
    for(const n of nodes){
      if(selectedSet.has(n.id)) continue;
      if(coreIds.has(n.id)){ unselectedCoreNodes.push(n); continue; }
      const links=selectedIds.filter(fid=>adj.get(fid)?.has(n.id));
      if(links.length>=2) sharedSelected.push(n);
      else if(links.length===1) directByCore.get(links[0]).push(n);
    }

    // Each selected ministry gets a circular local halo. Nodes shared by several selected ministries
    // are kept between the ministries so the shared context is immediately legible.
    for(const [fid,list] of directByCore){
      const cp=nodeMap.get(fid); if(!cp) continue;
      list.sort((a,b)=>hashString(a.id)-hashString(b.id));
      const perRing=Math.max(6,Math.min(10,Math.ceil(Math.sqrt(list.length+1)*2.2)));
      list.forEach((n,i)=>{
        const ring=Math.floor(i/perRing);
        const pos=i%perRing;
        const count=Math.min(perRing,list.length-ring*perRing);
        const rr=Math.min(145+ring*82,minDim*.40);
        const phase=(hashString(fid)%360)*Math.PI/180;
        const a=phase + (count===1?0:(pos/count)*Math.PI*2);
        const p=polar(cp.x,cp.y,rr,a);
        n.x=p.x; n.y=p.y;
      });
    }

    if(sharedSelected.length){
      let sx=0,sy=0; selectedNodes.forEach(n=>{sx+=n.x;sy+=n.y}); sx/=selectedNodes.length; sy/=selectedNodes.length;
      sharedSelected.sort((a,b)=>{const da=(a.name||'').localeCompare(b.name||'','ru'); return da||hashString(a.id)-hashString(b.id);});
      const perRing=Math.max(6,Math.min(10,Math.ceil(Math.sqrt(sharedSelected.length+1)*2.0)));
      sharedSelected.forEach((n,i)=>{
        const ring=Math.floor(i/perRing),pos=i%perRing,count=Math.min(perRing,sharedSelected.length-ring*perRing);
        const rr=Math.min(180+ring*78,minDim*.39);
        const a=-Math.PI/2+(pos/count)*Math.PI*2;
        const p=polar(sx,sy,rr,a); n.x=p.x;n.y=p.y;
      });
    }

    // Unselected core ministries are placed farther away, so they do not steal the visual center.
    const outerR=Math.min(315,minDim*.43);
    unselectedCoreNodes.forEach((n,i)=>{
      const a=-Math.PI/2+i*(Math.PI*2/Math.max(1,unselectedCoreNodes.length));
      const p=polar(cx,cy,outerR,a); n.x=p.x;n.y=p.y;
    });

    resolveNodeOverlaps(nodes,new Set(selectedIds),42,26);
  }
  resolveNodeOverlaps(nodes,new Set(CORE.map(([id])=>id)),34,18);
}

function resolveNodeOverlaps(nodes,fixedIds=new Set(),minGap=34,iterations=18){
  if(nodes.length<2)return;
  for(let it=0;it<iterations;it++){
    let moved=false;
    for(let i=0;i<nodes.length;i++){
      const a=nodes[i];
      for(let j=i+1;j<nodes.length;j++){
        const b=nodes[j];
        let dx=b.x-a.x,dy=b.y-a.y,dist=Math.hypot(dx,dy);
        if(dist<0.01){const h=hashString(a.id+'|'+b.id);dx=Math.cos((h%6283)/1000);dy=Math.sin((h%6283)/1000);dist=Math.hypot(dx,dy);}
        const target=Math.max(minGap,nodeVisualRadius(a)+nodeVisualRadius(b)+10);
        if(dist>=target)continue;
        const push=(target-dist)*0.56;const nx=dx/dist,ny=dy/dist;
        const af=!fixedIds.has(a.id),bf=!fixedIds.has(b.id);
        if(af&&bf){a.x-=nx*push/2;a.y-=ny*push/2;b.x+=nx*push/2;b.y+=ny*push/2;}
        else if(af){a.x-=nx*push;a.y-=ny*push;}
        else if(bf){b.x+=nx*push;b.y+=ny*push;}
        moved=true;
      }
    }
    if(!moved)break;
  }
}

function structureDepartmentHandoffLinks(ministryId){
  const ministryName=nodeById[ministryId]?.name||'';
  const deptRows=[...structureDeptById.values()].filter(r=>structureOrgId(r.Org)===ministryId);
  const deptIds=new Set(deptRows.map(r=>r.Dept_Global_ID));
  const rows=[];
  for(const arr of structureRolesByDept.values()) for(const r of arr) if(deptIds.has(r.Dept_Global_ID)) rows.push(r);
  const byProcess=new Map();
  for(const r of rows){const p=r.Process_ID||r.Process;if(!p)continue;if(!byProcess.has(p))byProcess.set(p,[]);byProcess.get(p).push(r);}
  const links=[];
  for(const [processId,recs] of byProcess){
    const owners=recs.filter(r=>String(r.Role||'').includes('A/R')||String(r.Basis||'').includes('TO_DEPT'));
    if(owners.length!==1)continue;
    const owner=owners[0];
    for(const sender of recs.filter(r=>String(r.Basis||'').includes('FROM_DEPT')&&r.Dept_Global_ID!==owner.Dept_Global_ID)){
      links.push({id:`STR:DEPTLINK:${sender.Dept_Global_ID}:${owner.Dept_Global_ID}:${processId}`,a:sender.Dept_Global_ID,b:owner.Dept_Global_ID,a_name:sender.Department,b_name:owner.Department,kind:'DEPARTMENT_HANDOFF',kind_label:'Междепартаментная процессная передача',direction:'DIRECTED_DERIVED',evidence:'EXPLICIT_CANONICAL_HANDOFF',org:ministryName,process_id:processId,process:sender.Process||processId,source:`FROM_DEPT · ${sender.Basis} | TO_DEPT · ${owner.Basis}`,source_records:[sender,owner],derived:true});
    }
  }
  for(const d of (SUPPLEMENTAL_DEPT_LINKS.links||[])){
    const a=d.from_dept_global_id,b=d.to_dept_global_id;
    if(d.org===ministryName && deptIds.has(a) && deptIds.has(b)){
      links.push({id:d.id,a,b,a_name:d.from_department,b_name:d.to_department,kind:(d.status==='CANDIDATE'||d.status==='WORKING_ANALYTICAL')?'DEPARTMENT_CANDIDATE':'DEPARTMENT_DOCUMENTED',kind_label:(d.status==='CANDIDATE'||d.status==='WORKING_ANALYTICAL')?'Аналитическая связь — требуется подтверждение':`Документированное взаимодействие (${d.document_year||'без даты'})`,direction:d.direction,evidence:d.evidence,org:d.org,source_title:d.source_title,source_url:d.source_url,document_year:d.document_year,basis:d.basis,current_structure_source:d.current_structure_source,current_structure_status:d.current_structure_status});
    }
  }

  // Contextual functional links: same canonical Process_ID in two departments.
  // These links show shared participation in a process, not a proven direct hand-off.
  const existingPairs=new Set(links.map(e=>[e.a,e.b].sort().join('|')));
  const procByDept=new Map(),recordsByDeptProc=new Map();
  for(const d of deptRows){procByDept.set(d.Dept_Global_ID,new Map());recordsByDeptProc.set(d.Dept_Global_ID,new Map());}
  const addProcessRecord=r=>{
    const did=r.Dept_Global_ID,pid=r.Process_ID||r.Process;
    if(!deptIds.has(did)||!pid)return;
    if(!procByDept.get(did).has(pid))procByDept.get(did).set(pid,r.Process||pid);
    if(!recordsByDeptProc.get(did).has(pid))recordsByDeptProc.get(did).set(pid,[]);
    recordsByDeptProc.get(did).get(pid).push(r);
  };
  for(const arr of structureProcessesByDept.values()) for(const r of arr) addProcessRecord(r);
  for(const arr of structureRolesByDept.values()) for(const r of arr) addProcessRecord(r);
  const ids=[...deptIds];
  for(let i=0;i<ids.length;i++) for(let j=i+1;j<ids.length;j++){
    const a=ids[i],b=ids[j],pair=[a,b].sort().join('|');
    if(existingPairs.has(pair))continue;
    const pa=procByDept.get(a)||new Map(),pb=procByDept.get(b)||new Map();
    const common=[...pa.keys()].filter(pid=>pb.has(pid));
    if(!common.length)continue;
    const names=[...new Set(common.map(pid=>pa.get(pid)||pb.get(pid)||pid))];
    const sourceRecords=[];
    for(const pid of common){sourceRecords.push(...(recordsByDeptProc.get(a)?.get(pid)||[]),...(recordsByDeptProc.get(b)?.get(pid)||[]));}
    links.push({
      id:`STR:SHAREDPROC:${a}:${b}:${common.map(x=>hashString(x)).join('-')}`,
      a,b,a_name:structureDepartmentName(a),b_name:structureDepartmentName(b),
      kind:'DEPARTMENT_SHARED_PROCESS',kind_label:common.length===1?'Общий процесс':'Общие процессы',
      direction:'UNDIRECTED_CONTEXTUAL',evidence:'SHARED_CANONICAL_PROCESS_ID',org:ministryName,
      process_ids:common,process_names:names,process:names.join(' · '),source_records:sourceRecords,derived_context:true
    });
  }
  return links;
}

function buildStructureGraph(){
  const nodes=[],links=[];structureNodeById.clear();
  const selectedMin=structureMinistryFocusId;
  const selectedDept=structureDeptDetailId&&structureDeptById.get(structureDeptDetailId);
  if(selectedMin){
    nodes.push({...nodeById[selectedMin],type:'MINISTRY',structure:true});
    for(const r of structureDeptById.values()) if(structureOrgId(r.Org)===selectedMin) nodes.push({id:r.Dept_Global_ID,name:r.Registry_Name,type:'DEPARTMENT',parentOrg:selectedMin,structure:true});
  } else {
    for(const [id,name] of CORE) nodes.push({...nodeById[id],id,name,type:'MINISTRY',structure:true});
  }
  for(const n of nodes) structureNodeById.set(n.id,n);
  // В выбранном министерстве принадлежность подразделений показывается пространственным контейнером,
  // поэтому радиальные ministry→department линии не рисуются. В обзорном режиме департаментов нет.
  if(!selectedMin){
    for(const n of nodes.filter(n=>n.type==='DEPARTMENT')) links.push({id:`STR:OWN:${n.id}`,a:n.parentOrg,b:n.id,a_name:nodeById[n.parentOrg]?.name||n.parentOrg,b_name:n.name,kind:'OWNERSHIP',kind_label:'Принадлежность подразделения',direction:'DIRECTED_EXPLICIT',evidence:structureDeptById.get(n.id)?.Registry_Evidence||'',source:structureDeptById.get(n.id)?.Registry_Source||'',org:structureDeptById.get(n.id)?.Org||'',source_urls:structureDeptById.get(n.id)?.Registry_Source||''});
  }
  // Все зафиксированные междепартаментные связи остаются видимыми внутри контейнера.
  if(selectedMin){
    const handoffLinks=structureDepartmentHandoffLinks(selectedMin);
    links.push(...handoffLinks.filter(e=>structureNodeById.has(e.a)&&structureNodeById.has(e.b)));
  }
  // Functions/processes remain a drill-down layer for the clicked department.
  if(selectedDept){
    const depId=selectedDept.Dept_Global_ID;
    const fnMap=new Map();
    for(const r of structureFunctionsByDept.get(depId)||[]){const fid=`STR:F:${depId}:${r.Function_ID||r.Function||hashString(JSON.stringify(r))}`;if(!fnMap.has(fid))fnMap.set(fid,r)}
    for(const [fid,r] of fnMap){const functionName=r.Function||r.Function_ID||'Функция';nodes.push({id:fid,name:`Функция: ${functionName}`,type:'FUNCTION',structure:true,sourceRecords:[r]});links.push({id:`STR:FUNC:${depId}:${fid}`,a:depId,b:fid,a_name:selectedDept.Registry_Name,b_name:functionName,kind:'FUNCTION',kind_label:'Связь с функцией',direction:'DIRECTED_EXPLICIT',evidence:r.Evidence||r.Confidence||'',source:r.Source||r.Basis||''});structureNodeById.set(fid,nodes[nodes.length-1]);}
    const procMap=new Map();
    for(const r of structureProcessesByDept.get(depId)||[]){const pid=`STR:P:${depId}:${r.Process_ID||r.Process||hashString(JSON.stringify(r))}`;if(!procMap.has(pid))procMap.set(pid,[]);procMap.get(pid).push(r)}
    for(const r of structureRolesByDept.get(depId)||[]){const pid=`STR:P:${depId}:${r.Process_ID||r.Process||hashString(JSON.stringify(r))}`;if(!procMap.has(pid))procMap.set(pid,[]);procMap.get(pid).push(r)}
    for(const [pid,records] of procMap){const primary=records.find(r=>r.Process||r.Process_ID)||records[0];const processName=primary?.Process||primary?.Process_ID||'Процесс';const evidence=[...new Set(records.map(r=>r.Evidence||r.Confidence||'').filter(Boolean))].join(' · ');const source=[...new Set(records.map(r=>r.Source||r.Basis||r.Role_Detail||'').filter(Boolean))].join(' · ');nodes.push({id:pid,name:`Процесс: ${processName}`,type:'PROCESS',structure:true,sourceRecords:records});links.push({id:`STR:PROC:${depId}:${pid}`,a:depId,b:pid,a_name:selectedDept.Registry_Name,b_name:processName,kind:'PROCESS',kind_label:'Связь с процессом',direction:'DIRECTED_EXPLICIT',evidence:evidence,source:source});structureNodeById.set(pid,nodes[nodes.length-1]);}
  }
  return {nodes,links};
}
function structureMinistryAreaRadius(){
  const minDim=Math.min(W,H);
  return Math.max(255,Math.min(390,minDim*.43));
}
function arrangeStructureLayout(nodes,links){
  const cx=W/2,cy=H/2,minDim=Math.min(W,H);const nm=new Map(nodes.map(n=>[n.id,n]));
  const ministries=nodes.filter(n=>n.type==='MINISTRY'),depts=nodes.filter(n=>n.type==='DEPARTMENT'),funcs=nodes.filter(n=>n.type==='FUNCTION'),procs=nodes.filter(n=>n.type==='PROCESS');
  if(structureMinistryFocusId){
    const m=nm.get(structureMinistryFocusId);if(!m)return;m.x=cx;m.y=cy;
    const ds=depts.filter(n=>n.parentOrg===structureMinistryFocusId);
    const linkDegree=new Map(ds.map(n=>[n.id,0]));
    for(const e of links){if(linkDegree.has(e.a)&&linkDegree.has(e.b)){linkDegree.set(e.a,(linkDegree.get(e.a)||0)+1);linkDegree.set(e.b,(linkDegree.get(e.b)||0)+1)}}
    ds.sort((a,b)=>(linkDegree.get(b.id)||0)-(linkDegree.get(a.id)||0)||a.name.localeCompare(b.name,'ru'));
    const R=structureMinistryAreaRadius();
    const rings=[{cap:7,r:R*.31},{cap:11,r:R*.56},{cap:16,r:R*.78},{cap:22,r:R*.91}];
    let idx=0;
    for(let ri=0;ri<rings.length&&idx<ds.length;ri++){
      const ring=rings[ri],count=Math.min(ring.cap,ds.length-idx),offset=(ri%2)*Math.PI/Math.max(1,count);
      for(let k=0;k<count;k++,idx++){
        const n=ds[idx],a=-Math.PI/2+offset+(k/count)*Math.PI*2,p=polar(cx,cy,ring.r,a);
        n.x=p.x;n.y=p.y;n.fx=null;n.fy=null;
      }
    }
    // Rare overflow: keep extra departments in a compact inner ring rather than outside the ministry.
    for(;idx<ds.length;idx++){
      const n=ds[idx],k=idx-rings.reduce((s,r)=>s+r.cap,0),count=Math.max(1,ds.length-idx),a=-Math.PI/2+(k/Math.max(1,count))*Math.PI*2,p=polar(cx,cy,R*.16,a);n.x=p.x;n.y=p.y;n.fx=null;n.fy=null;
    }
    if(structureDeptDetailId){
      const dep=nm.get(structureDeptDetailId);
      if(dep){
        const ax=Math.atan2(dep.y-cy,dep.x-cx),fr=R+82,prr=R+158;
        funcs.forEach((n,i)=>{const span=Math.PI*.72,a=ax-span/2+(i/Math.max(1,funcs.length-1))*span,p=polar(cx,cy,fr,a);n.x=p.x;n.y=p.y;n.fx=null;n.fy=null});
        procs.forEach((n,i)=>{const span=Math.PI*.88,a=ax-span/2+(i/Math.max(1,procs.length-1))*span,p=polar(cx,cy,prr,a);n.x=p.x;n.y=p.y;n.fx=null;n.fy=null});
      }
    }
    return;
  }
  // Default structure overview: only the nine major ministries. Departments are revealed by ministry selection.
  const radius=Math.min(285,minDim*.36);ministries.forEach((m,i)=>{const a=-Math.PI/2+i*(Math.PI*2/Math.max(1,ministries.length)),p=polar(cx,cy,radius,a);m.x=p.x;m.y=p.y;m.fx=null;m.fy=null});
  resolveNodeOverlaps(nodes,new Set(),40,36);
}
function structureLabelLines(name,type,id){
  const raw=String(name||'');
  if(type==='DEPARTMENT'){
    const words=raw.split(/\s+/),lines=[];let line='';const maxChars=24;
    for(const w of words){
      const next=line?line+' '+w:w;
      if(next.length>maxChars&&line){lines.push(line);line=w}else line=next;
      if(lines.length===2&&line.length>maxChars+5){lines.push(line.slice(0,maxChars+4).trimEnd()+'…');line='';break}
    }
    if(line)lines.push(line);
    if(lines.length>3){lines.length=3;lines[2]=lines[2].slice(0,maxChars+3).trimEnd()+'…'}
    return lines.slice(0,3);
  }
  return [raw.length<=38?raw:raw.slice(0,37).trimEnd()+'…'];
}

function structureSourceIdsForRecords(org, records) {
  const ids = new Set();
  const codes = [];
  const urls = [];
  for (const record of records || []) {
    for (const value of String(record.Source_ID || record.source_id || '').split(/[;,]/)) if (value.trim()) ids.add(value.trim());
    for (const value of String(record.Source || '').split(/[;,]/)) if (value.trim()) codes.push(value.trim());
    for (const value of String(record.Registry_Source || '').split('|')) if (value.trim()) urls.push(value.trim());
  }
  for (const source of structureSourceRecordsForCodes(org, codes)) ids.add(source.source_id);
  for (const url of urls) {
    const source = DATA.sources.find(item => item.url === url);
    if (source) ids.add(source.source_id);
  }
  return [...ids];
}

function structureDepartmentHandoffLinks(ministryId) {
  const ministryName = nodeById[ministryId]?.name || '';
  const deptRows = [...structureDeptById.values()].filter(record => structureOrgId(record.Org) === ministryId);
  const deptIds = new Set(deptRows.map(record => record.Dept_Global_ID));
  const rows = [];
  for (const values of structureRolesByDept.values()) for (const record of values) if (deptIds.has(record.Dept_Global_ID)) rows.push(record);
  const byProcess = new Map();
  for (const record of rows) {
    const processId = record.Process_ID || record.Process;
    if (!processId) continue;
    if (!byProcess.has(processId)) byProcess.set(processId, []);
    byProcess.get(processId).push(record);
  }
  const links = [];
  for (const [processId, records] of byProcess) {
    const owners = records.filter(record => String(record.Role || '').includes('A/R') || String(record.Basis || '').includes('TO_DEPT'));
    if (owners.length !== 1) continue;
    const owner = owners[0];
    for (const sender of records.filter(record => String(record.Basis || '').includes('FROM_DEPT') && record.Dept_Global_ID !== owner.Dept_Global_ID)) {
      const sourceRecords = [sender, owner];
      const sourceIds = structureSourceIdsForRecords(ministryName, sourceRecords);
      if (!sourceIds.length) continue;
      links.push({ id: `STR:DEPTLINK:${sender.Dept_Global_ID}:${owner.Dept_Global_ID}:${processId}`, a: sender.Dept_Global_ID, b: owner.Dept_Global_ID, a_name: sender.Department, b_name: owner.Department, kind: 'DEPARTMENT_HANDOFF', kind_label: 'Междепартаментная процессная передача', direction: 'DIRECTED_DERIVED', evidence: sourceIds.length ? 'EXPLICIT_CANONICAL_HANDOFF_WITH_SOURCE' : 'EXPLICIT_CANONICAL_HANDOFF_SOURCE_GAP', org: ministryName, process_id: processId, process: sender.Process || processId, source: `FROM_DEPT · ${sender.Basis} | TO_DEPT · ${owner.Basis}`, source_records: sourceRecords, source_ids: sourceIds, derived: true });
    }
  }
  for (const record of SUPPLEMENTAL_DEPT_LINKS.links || []) {
    const a = record.from_dept_global_id, b = record.to_dept_global_id;
    if (record.org !== ministryName || !deptIds.has(a) || !deptIds.has(b)) continue;
    const candidate = record.status === 'CANDIDATE' || record.status === 'WORKING_ANALYTICAL';
    const generalDocumented = record.documentation_scope === 'GENERAL_ORGANIZATIONAL';
    links.push({ ...record, id: record.id, a, b, a_name: record.from_department, b_name: record.to_department, kind: candidate ? 'DEPARTMENT_CANDIDATE' : 'DEPARTMENT_DOCUMENTED', kind_label: candidate ? 'Аналитическая функциональная связь — требуется прямое подтверждение' : (generalDocumented ? 'Документированное общее организационное взаимодействие' : 'Документированное междепартаментное взаимодействие'), direction: record.direction, evidence: record.evidence, org: record.org, process: record.process || record.process_id || '', source_ids: String(record.source_id || '').split(/[;,]/).map(value => value.trim()).filter(Boolean), source_url: record.source_url, source_title: record.source_title });
  }

  const existingPairs = new Set(links.map(edge => [edge.a, edge.b].sort().join('|')));
  const processesByDept = new Map(), recordsByDeptProcess = new Map();
  for (const department of deptRows) {
    processesByDept.set(department.Dept_Global_ID, new Map());
    recordsByDeptProcess.set(department.Dept_Global_ID, new Map());
  }
  const addProcessRecord = record => {
    const departmentId = record.Dept_Global_ID, processId = record.Process_ID || record.Process;
    if (!deptIds.has(departmentId) || !processId) return;
    if (!processesByDept.get(departmentId).has(processId)) processesByDept.get(departmentId).set(processId, record.Process || processId);
    if (!recordsByDeptProcess.get(departmentId).has(processId)) recordsByDeptProcess.get(departmentId).set(processId, []);
    recordsByDeptProcess.get(departmentId).get(processId).push(record);
  };
  for (const values of structureProcessesByDept.values()) for (const record of values) addProcessRecord(record);
  for (const values of structureRolesByDept.values()) for (const record of values) addProcessRecord(record);
  const ids = [...deptIds];
  for (let i = 0; i < ids.length; i++) for (let j = i + 1; j < ids.length; j++) {
    const a = ids[i], b = ids[j], pair = [a, b].sort().join('|');
    if (existingPairs.has(pair)) continue;
    const left = processesByDept.get(a) || new Map(), right = processesByDept.get(b) || new Map();
    const common = [...left.keys()].filter(processId => right.has(processId));
    if (!common.length) continue;
    const sourceRecords = [];
    for (const processId of common) sourceRecords.push(...(recordsByDeptProcess.get(a)?.get(processId) || []), ...(recordsByDeptProcess.get(b)?.get(processId) || []));
    const sourceIds = structureSourceIdsForRecords(ministryName, sourceRecords);
    if (!sourceIds.length) continue;
    const names = [...new Set(common.map(processId => left.get(processId) || right.get(processId) || processId))];
    links.push({ id: `STR:SHAREDPROC:${a}:${b}:${common.map(value => hashString(value)).join('-')}`, a, b, a_name: structureDepartmentName(a), b_name: structureDepartmentName(b), kind: 'DEPARTMENT_SHARED_PROCESS', kind_label: common.length === 1 ?   }
  return links;
}

function structureMinistryInteractionLinks() {
  const groups = new Map();
  for (const edge of DATA.edges) {
    if (edge.a === edge.b || edge.channel_binding || edge.source_coverage === 'EXCLUDED_NO_DOCUMENT' || !edge.source_ids?.length) continue;
    if (!coreIds.has(edge.a) || !coreIds.has(edge.b)) continue;
    const pair = [edge.a, edge.b].sort();
    const key = pair.join('|');
    if (!groups.has(key)) groups.set(key, { a: pair[0], b: pair[1], edges: [] });
    groups.get(key).edges.push(edge);
  }
  return [...groups.values()].map(group => {
    const sourceIds = [...new Set(group.edges.flatMap(edge => edge.source_ids || []))];
    return { id: `STR:MINISTRY:${group.a}:${group.b}`, a: group.a, b: group.b, a_name: nodeById[group.a]?.name || group.a, b_name: nodeById[group.b]?.name || group.b, kind: 'MINISTRY_INTERACTION', kind_label: 'Межведомственное взаимодействие основных органов', direction: 'UNDIRECTED_AGGREGATED', evidence: sourceIds.length ? 'SOURCE_BACKED_AGGREGATION' : 'SOURCE_GAP', org: 'Межведомственный контур', source_ids: sourceIds, base_edge_ids: group.edges.map(edge => edge.id), relation_types: [...new Set(group.edges.flatMap(edge => String(edge.relation_types || '').split(';').map(value => value.trim()).filter(Boolean)))], channels: [...new Set(group.edges.flatMap(edge => channelNorm(edge.channels)))], processes: [...new Set(group.edges.flatMap(edge => String(edge.processes || '').split('|').map(value => value.trim()).filter(Boolean)))], evidence_count: group.edges.reduce((sum, edge) => sum + Number(edge.evidence_count || 0), 0) };

function buildStructureGraph() {
  const nodes = [], links = [];
  structureNodeById.clear();
  const selectedMin = structureMinistryFocusId;
  const selectedDept = structureDeptDetailId && structureDeptById.get(structureDeptDetailId);
  const ministryIds = selectedMin ? [selectedMin] : CORE.map(([id]) => id);
  for (const ministryId of ministryIds) nodes.push({ ...nodeById[ministryId], id: ministryId, name: nodeById[ministryId]?.name || CORE.find(([id]) => id === ministryId)?.[1] || ministryId, type: 'MINISTRY', structure: true });
  for (const record of structureDeptById.values()) {
    const parentOrg = structureOrgId(record.Org);
    if (ministryIds.includes(parentOrg)) nodes.push({ id: record.Dept_Global_ID, name: record.Registry_Name, type: 'DEPARTMENT', parentOrg, structure: true });
  }
  for (const node of nodes) structureNodeById.set(node.id, node);
  for (const ministryId of ministryIds) links.push(...structureDepartmentHandoffLinks(ministryId).filter(edge => structureNodeById.has(edge.a) && structureNodeById.has(edge.b)));
  if (!selectedMin) links.push(...structureMinistryInteractionLinks());
  if (selectedDept) {
    const departmentId = selectedDept.Dept_Global_ID;
    const functions = new Map();
    for (const record of structureFunctionsByDept.get(departmentId) || []) {
      const id = `STR:F:${departmentId}:${record.Function_ID || record.Function || hashString(JSON.stringify(record))}`;
      if (!functions.has(id)) functions.set(id, record);
    }
    for (const [id, record] of functions) {
      const name = record.Function || record.Function_ID || 'Функция';
      const node = { id, name: `Функция: ${name}`, type: 'FUNCTION', structure: true, sourceRecords: [record] };
      nodes.push(node); structureNodeById.set(id, node);
      links.push({ id: `STR:FUNC:${departmentId}:${id}`, a: departmentId, b: id, a_name: selectedDept.Registry_Name, b_name: name, kind: 'FUNCTION', kind_label: 'Связь с функцией', direction: 'DIRECTED_EXPLICIT', evidence: record.Evidence || record.Confidence || '', source: record.Source || record.Basis || '' });
    }
    const processes = new Map();
    for (const record of [...(structureProcessesByDept.get(departmentId) || []), ...(structureRolesByDept.get(departmentId) || [])]) {
      const id = `STR:P:${departmentId}:${record.Process_ID || record.Process || hashString(JSON.stringify(record))}`;
      if (!processes.has(id)) processes.set(id, []);
      processes.get(id).push(record);
    }
    for (const [id, records] of processes) {
      const primary = records.find(record => record.Process || record.Process_ID) || records[0];
      const name = primary?.Process || primary?.Process_ID || 'Процесс';
      const node = { id, name: `Процесс: ${name}`, type: 'PROCESS', structure: true, sourceRecords: records };
      nodes.push(node); structureNodeById.set(id, node);
      links.push({ id: `STR:PROC:${departmentId}:${id}`, a: departmentId, b: id, a_name: selectedDept.Registry_Name, b_name: name, kind: 'PROCESS', kind_label: 'Связь с процессом', direction: 'DIRECTED_EXPLICIT', evidence: [...new Set(records.map(record => record.Evidence || record.Confidence || '').filter(Boolean))].join(' · '), source: [...new Set(records.map(record => record.Source || record.Basis || record.Role_Detail || '').filter(Boolean))].join(' · ') });
    }
  }
  return { nodes, links };
}

function structureNodeRadius(node) {
  if (node.type === 'MINISTRY') return structureMinistryFocusId ? 18 : 12;
  if (node.type === 'DEPARTMENT') return structureMinistryFocusId ? 10 : 4.5;
  if (node.type === 'FUNCTION' || node.type === 'PROCESS') return 8;
  return 7;
}

function structureOverviewAreaRadius() {
  const columns = 4, rows = Math.ceil(CORE.length / columns);
  return Math.max(64, Math.min(W / columns, H / rows) * .36);
}

function arrangeStructureLayout(nodes, links) {
  const cx = W / 2, cy = H / 2;
  const nodeMap = new Map(nodes.map(node => [node.id, node]));
  const ministries = nodes.filter(node => node.type === 'MINISTRY');
  const departments = nodes.filter(node => node.type === 'DEPARTMENT');
  const functions = nodes.filter(node => node.type === 'FUNCTION');
  const processes = nodes.filter(node => node.type === 'PROCESS');
  if (structureMinistryFocusId) {
    const ministry = nodeMap.get(structureMinistryFocusId);
    if (!ministry) return;
    ministry.x = cx; ministry.y = cy;
    const ownDepartments = departments.filter(node => node.parentOrg === structureMinistryFocusId);
    const degree = new Map(ownDepartments.map(node => [node.id, 0]));
    for (const edge of links) if (degree.has(edge.a) && degree.has(edge.b)) { degree.set(edge.a, degree.get(edge.a) + 1); degree.set(edge.b, degree.get(edge.b) + 1); }
    ownDepartments.sort((a, b) => degree.get(b.id) - degree.get(a.id) || a.name.localeCompare(b.name, 'ru'));
    const radius = structureMinistryAreaRadius();
    const rings = [{ cap: 7, r: radius * .31 }, { cap: 11, r: radius * .56 }, { cap: 16, r: radius * .78 }, { cap: 30, r: radius * .91 }];
    let index = 0;
    for (let ringIndex = 0; ringIndex < rings.length && index < ownDepartments.length; ringIndex++) {
      const ring = rings[ringIndex], count = Math.min(ring.cap, ownDepartments.length - index), offset = ringIndex % 2 * Math.PI / Math.max(1, count);
      for (let item = 0; item < count; item++, index++) { const node = ownDepartments[index], point = polar(cx, cy, ring.r, -Math.PI / 2 + offset + item / count * Math.PI * 2); node.x = point.x; node.y = point.y; }
    }
    if (structureDeptDetailId) {
      const department = nodeMap.get(structureDeptDetailId);
      if (department) {
        const axis = Math.atan2(department.y - cy, department.x - cx), functionRadius = radius + 82, processRadius = radius + 158;
        functions.forEach((node, index) => { const span = Math.PI * .72, point = polar(cx, cy, functionRadius, axis - span / 2 + index / Math.max(1, functions.length - 1) * span); node.x = point.x; node.y = point.y; });
        processes.forEach((node, index) => { const span = Math.PI * .88, point = polar(cx, cy, processRadius, axis - span / 2 + index / Math.max(1, processes.length - 1) * span); node.x = point.x; node.y = point.y; });
      }
    }
    return;
  }
  const columns = 4, rows = Math.ceil(ministries.length / columns), cellWidth = W / columns, cellHeight = H / rows, radius = structureOverviewAreaRadius();
  ministries.forEach((ministry, index) => {
    const column = index % columns, row = Math.floor(index / columns);
    ministry.x = cellWidth * (column + .5); ministry.y = cellHeight * (row + .5);
    const ownDepartments = departments.filter(node => node.parentOrg === ministry.id).sort((a, b) => a.name.localeCompare(b.name, 'ru'));
    const rings = ownDepartments.length <= 14 ? [{ cap: 14, r: radius * .72 }] : [{ cap: 12, r: radius * .53 }, { cap: 30, r: radius * .84 }];
    let cursor = 0;
    for (let ringIndex = 0; ringIndex < rings.length && cursor < ownDepartments.length; ringIndex++) {
      const ring = rings[ringIndex], count = Math.min(ring.cap, ownDepartments.length - cursor), offset = ringIndex * Math.PI / Math.max(1, count);
      for (let item = 0; item < count; item++, cursor++) { const node = ownDepartments[cursor], point = polar(ministry.x, ministry.y, ring.r, -Math.PI / 2 + offset + item / count * Math.PI * 2); node.x = point.x; node.y = point.y; }
    }
  });
}

function structureEdgeDetail(edge) {
  const arrow = String(edge.direction || '').startsWith('UNDIRECTED') || edge.kind === 'DEPARTMENT_SHARED_PROCESS' || edge.kind === 'MINISTRY_INTERACTION' ? '↔' : '→';
  const status = edge.kind === 'DEPARTMENT_CANDIDATE' ? 'Аналитический функциональный контекст; требуется прямой регламент или документ передачи.' : edge.kind === 'DEPARTMENT_DOCUMENTED' ? 'Документированная связь из исходной матрицы.' : edge.kind === 'MINISTRY_INTERACTION' ? `Агрегировано исходных связей: ${(edge.base_edge_ids || []).length}.` : edge.kind === 'DEPARTMENT_SHARED_PROCESS' ? 'Совместное участие в подтверждённом процессе; прямой обмен не утверждается.' : 'Связь структурного слоя.';
  const details = edge.kind === 'MINISTRY_INTERACTION' ? `<p><strong>Типы:</strong> ${esc((edge.relation_types || []).join(' · ') || '—')}</p><p><strong>Каналы:</strong> ${esc((edge.channels || []).join(' · ') || '—')}</p>` : `<p><strong>Процесс / основание:</strong> ${esc(edge.process || edge.basis || edge.source || '—')}</p>`;
  return `<h2>${esc(edge.a_name)} ${arrow} ${esc(edge.b_name)}</h2><div class="meta">${esc(edge.kind_label || edge.kind || 'Связь')}</div><div class="card"><p><strong>Статус:</strong> ${esc(status)}</p><p><strong>Доказательность:</strong> ${esc(edge.evidence || '—')}</p>${details}${edge.input_data ? `<p><strong>Вход:</strong> ${esc(edge.input_data)}</p>` : ''}${edge.output_data ? `<p><strong>Выход:</strong> ${esc(edge.output_data)}</p>` : ''}${edge.qa_note ? `<p class="small">${esc(edge.qa_note)}</p>` : ''}</div>${structureSourceHtml(edge)}`;
}

  arrangeStructureLayout(state.nodes, state.links);
  edgesLayer.innerHTML = ''; nodesLayer.innerHTML = '';
  const ministryIds = structureMinistryFocusId ? [structureMinistryFocusId] : CORE.map(([id]) => id);
  for (const ministryId of ministryIds) {
    const ministry = state.nodes.find(node => node.id === ministryId);
    if (!ministry) continue;
    const radius = structureMinistryFocusId ? structureMinistryAreaRadius() : structureOverviewAreaRadius();
    const color = coreColor(ministryId);
    const ownDepartments = state.nodes.filter(node => node.type === 'DEPARTMENT' && node.parentOrg === ministryId);
    const internal = state.links.filter(edge => edge.kind !== 'MINISTRY_INTERACTION' && ownDepartments.some(node => node.id === edge.a) && ownDepartments.some(node => node.id === edge.b));
    const boundary = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    boundary.classList.add('structure-ministry-boundary'); boundary.setAttribute('cx', ministry.x); boundary.setAttribute('cy', ministry.y); boundary.setAttribute('r', radius); boundary.setAttribute('fill', color); boundary.setAttribute('fill-opacity', structureMinistryFocusId ? '.075' : '.045'); boundary.setAttribute('stroke', color); boundary.setAttribute('stroke-opacity', '.58'); boundary.setAttribute('stroke-width', structureMinistryFocusId ? '2.3' : '1.5');
    boundary.addEventListener('click', event => { event.stopPropagation(); focusStructureMinistry(ministryId); }); edgesLayer.appendChild(boundary);
    const title = document.createElementNS('http://www.w3.org/2000/svg', 'text'); title.classList.add('structure-ministry-boundary-title'); title.setAttribute('x', ministry.x); title.setAttribute('y', ministry.y - radius + (structureMinistryFocusId ? 30 : 15)); title.setAttribute('fill', color); title.textContent = ministry.name; nodesLayer.appendChild(title);
    const meta = document.createElementNS('http://www.w3.org/2000/svg', 'text'); meta.classList.add('structure-ministry-boundary-meta'); meta.setAttribute('x', ministry.x); meta.setAttribute('y', ministry.y - radius + (structureMinistryFocusId ? 48 : 28)); meta.textContent = `${ownDepartments.length} подразделений · ${internal.length} связей`; nodesLayer.appendChild(meta);
  }
  for (const edge of state.links) {
    for (const hitMode of [true, false]) {
      const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
      line.classList.add(hitMode ? 'edge-hit' : 'edge', 'structure-edge');
      if (edge.kind === 'DEPARTMENT_HANDOFF') line.classList.add('department-handoff');
      if (edge.kind === 'DEPARTMENT_DOCUMENTED') line.classList.add('department-documented');
      if (edge.kind === 'DEPARTMENT_CANDIDATE') line.classList.add('department-candidate');
      if (edge.kind === 'DEPARTMENT_SHARED_PROCESS') line.classList.add('department-shared-process');
      if (edge.kind === 'MINISTRY_INTERACTION') line.classList.add('ministry-interaction');
      if (edge.kind === 'FUNCTION') line.classList.add('function-edge');
      if (edge.kind === 'PROCESS') line.classList.add('process-edge');
      line.dataset.id = edge.id;
      if (hitMode) { line.addEventListener('mouseenter', event => structureEdgeTooltip(edge, event)); line.addEventListener('mousemove', moveTooltip); line.addEventListener('mouseleave', hideTooltip); line.addEventListener('click', event => { event.stopPropagation(); selectedEdge = edge; selectedNode = null; detail.innerHTML = structureEdgeDetail(edge); }); }
      edgesLayer.appendChild(line);
    }
  }
  for (const node of state.nodes) {
    if (structureMinistryFocusId && node.type === 'MINISTRY') continue;
    const group = document.createElementNS('http://www.w3.org/2000/svg', 'g'); group.classList.add('node'); group.dataset.id = node.id;
    if (node.type === 'MINISTRY') group.classList.add('structure-ministry', 'ministry', 'core-ministry');
    if (node.type === 'DEPARTMENT') group.classList.add('structure-dept');
    if (node.type === 'FUNCTION') group.classList.add('structure-function');
    if (node.type === 'PROCESS') group.classList.add('structure-process');
    const radius = structureNodeRadius(node), circle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
    circle.setAttribute('r', radius); circle.setAttribute('fill', node.type === 'MINISTRY' ? coreColor(node.id) : node.type === 'DEPARTMENT' ? coreColor(node.parentOrg) : colorForType(node.type));
    if (node.type === 'DEPARTMENT') circle.setAttribute('fill-opacity', structureMinistryFocusId ? '.88' : '.78');
    group.appendChild(circle);
    const title = document.createElementNS('http://www.w3.org/2000/svg', 'title'); title.textContent = node.name; group.appendChild(title);
    if (structureMinistryFocusId || (node.type !== 'DEPARTMENT' && node.type !== 'MINISTRY')) {
      const text = document.createElementNS('http://www.w3.org/2000/svg', 'text'); text.setAttribute('x', radius + 7); text.setAttribute('y', 4);
      for (const [index, lineText] of structureLabelLines(node.name, node.type, node.id).entries()) { const span = document.createElementNS('http://www.w3.org/2000/svg', 'tspan'); span.textContent = lineText; span.setAttribute('x', node.type === 'MINISTRY' ? 0 : radius + 7); span.setAttribute('dy', index === 0 ? '0' : node.type === 'DEPARTMENT' ? '13' : '12'); text.appendChild(span); }
      if (node.type === 'MINISTRY') text.setAttribute('font-weight', '700'); group.appendChild(text); updateStructureLabel(group, node, radius);
    }
    group.addEventListener('click', event => { event.stopPropagation(); if (node.type === 'DEPARTMENT') focusDepartment(node.id); else if (node.type === 'MINISTRY') focusStructureMinistry(node.id); else selectNode(node.id); });
    group.addEventListener('mousedown', event => startNodeDrag(event, node)); nodesLayer.appendChild(group);
  }
  drawStructurePositions(); updateStats();
}

function ministryContextCategory(node,otherId){
  if(coreIds.has(otherId))return {key:'CORE',label:'Другие основные ведомства'};
  if(isMinistryNode(node))return {key:'MINISTRIES',label:'Другие министерства'};
  const t=node?.type||'UNKNOWN';
  if(['MINISTRY','FEDERAL_SERVICE','FEDERAL_AGENCY','REGIONAL_AUTHORITY','MUNICIPALITY','CORPORATION','INSTITUTION','ORGANIZATION','ACTOR_GROUP','COMPOSITE_ACTOR'].includes(t))return {key:'ORGS',label:'Организации и органы власти'};
  if(['SYSTEM','SYSTEM_ARTIFACT'].includes(t))return {key:'SYSTEMS',label:'Информационные системы'};
  if(t==='REGISTER')return {key:'REGISTERS',label:'Реестры'};
  if(t==='CHANNEL')return {key:'CHANNELS',label:'Каналы связи'};
  if(t==='DATA_CLASS')return {key:'DATA',label:'Данные'};
  if(['PROCESS','FUNCTION','FUNCTIONAL_BLOCK','RESULT'].includes(t))return {key:'PROCESSES',label:'Процессы, функции и результаты'};
  if(t==='DEPARTMENT')return {key:'DEPARTMENTS',label:'Подразделения'};
  return {key:'OTHER',label:'Прочие сущности'};
}
