  function internalDepartmentLabelLines(name,dense=false){
    const base=String(name||'').replace(/^Департамент\s+/i,'').replace(/^Управление\s+/i,'Управление ');
    const words=base.split(/\s+/).filter(Boolean),lines=[]; let line='';
    const maxChars=dense?16:18, maxLines=2;
    for (const word of words) {
      const next=line?`${line} ${word}`:word;
      if (next.length>maxChars && line) {
        lines.push(line); line=word;
      } else line=next;
      if (lines.length===maxLines-1 && line.length>maxChars+5) break;
    }
    if (line) lines.push(line);
    const out=[];
    let rest=lines.join(' ').trim();
    while (rest && out.length<maxLines) {
      if (rest.length<=maxChars) { out.push(rest); rest=''; break; }
      let cut=rest.lastIndexOf(' ',maxChars);
      if (cut<Math.floor(maxChars*0.55)) cut=maxChars;
      out.push(rest.slice(0,cut).trim());
      rest=rest.slice(cut).trim();
    }
    if (rest) out[Math.max(0,out.length-1)]=out[Math.max(0,out.length-1)].replace(/[.,;:!?]?$/,'')+'…';
    return out.filter(Boolean);
  }

<script id="ui-node-search-patch-20261003">
(function(){
  const root=document.getElementById('atlasNodeSearch');
  const input=document.getElementById('atlasNodeSearchInput');
  const results=document.getElementById('atlasNodeSearchResults');
  const clear=document.getElementById('atlasNodeSearchClear');
  if(!root||!input||!results||!clear||typeof DATA==='undefined'||typeof nodeById==='undefined')return;

  const normalize=value=>String(value||'').toLocaleLowerCase('ru').replace(/ё/g,'е').replace(/\s+/g,' ').trim();
  const searchableNodes=DATA.nodes.filter(n=>n&&n.id&&n.name&&(
    (typeof ORGANIZATION_TYPES!=='undefined'&&ORGANIZATION_TYPES.has(n.type)) ||
    (typeof coreIds!=='undefined'&&coreIds.has(n.id))
  ));
  let current=[];
  let activeIndex=-1;

  function nodeTypeLabel(n){
    try{return typeof labelType==='function'?labelType(n.type):(n.type||'узел');}
    catch(_){return n.type||'узел';}
  }
  function scoreNode(n,q){
    const name=normalize(n.name);
    if(name===q)return 0;
    if(name.startsWith(q))return 1;
    const wordStart=name.split(/[\s/()«»"'—–-]+/).some(part=>part.startsWith(q));
    if(wordStart)return 2;
    if(name.includes(q))return 3;
    const aliases=[].concat(n.aliases||[],n.canonical_alias_names||[],n.minfin_aliases||[]).map(normalize);
    if(aliases.some(a=>a===q))return 4;
    if(aliases.some(a=>a.startsWith(q)))return 5;
    if(aliases.some(a=>a.includes(q)))return 6;
    return 99;
  }
  function findMatches(q){
    if(!q)return [];
    return searchableNodes.map(n=>({n,score:scoreNode(n,q)})).filter(x=>x.score<99)
      .sort((a,b)=>a.score-b.score||a.n.name.localeCompare(b.n.name,'ru')).slice(0,12).map(x=>x.n);
  }
  function closeResults(){results.hidden=true;input.setAttribute('aria-expanded','false');activeIndex=-1;}
  function setActive(index){
    const buttons=[...results.querySelectorAll('.atlas-node-search-option')];
    if(!buttons.length){activeIndex=-1;return;}
    activeIndex=Math.max(0,Math.min(index,buttons.length-1));
    buttons.forEach((b,i)=>b.classList.toggle('active',i===activeIndex));
    buttons[activeIndex]?.scrollIntoView({block:'nearest'});
  }
  function escapeHtml(value){
    return String(value??'').replace(/[&<>"']/g,ch=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[ch]));
  }
  function render(){
    const q=normalize(input.value);
    if(!q){closeResults();results.innerHTML='';return;}
    current=findMatches(q);
    activeIndex=-1;
    if(!current.length){
      results.innerHTML='<div class="atlas-node-search-empty">Совпадений среди организационных узлов не найдено.</div>';
      results.hidden=false;input.setAttribute('aria-expanded','true');return;
    }
    results.innerHTML=current.map((n,i)=>`<button type="button" class="atlas-node-search-option" role="option" data-node-search-index="${i}"><span class="atlas-node-search-name">${escapeHtml(n.name)}</span><span class="atlas-node-search-type">${escapeHtml(nodeTypeLabel(n))}</span></button>`).join('');
    results.hidden=false;input.setAttribute('aria-expanded','true');
    results.querySelectorAll('.atlas-node-search-option').forEach(button=>button.addEventListener('click',()=>choose(Number(button.dataset.nodeSearchIndex))));
  }
  function choose(index){
    const n=current[index];if(!n)return;
    input.value=n.name;closeResults();
    const legacy=document.getElementById('search');if(legacy){legacy.value='';if(typeof applySearch==='function')applySearch();}
    try{
      if(typeof setGraphMode==='function')setGraphMode('OVERVIEW');
      if(typeof focusExternalNode==='function')focusExternalNode(n.id);
      else if(typeof setFocus==='function')setFocus(n.id);
      else if(typeof selectNode==='function')selectNode(n.id);
      document.body.classList.add('detail-open');
      const toggle=document.getElementById('detailToggle');if(toggle)toggle.setAttribute('aria-expanded','true');
      requestAnimationFrame(()=>{
        try{if(typeof selectNode==='function')selectNode(n.id);}catch(_){}
        const selected=document.querySelector(`#nodesLayer .node[data-id="${CSS.escape(n.id)}"]`);
        if(selected){selected.classList.add('selected');}
      });
    }catch(err){console.error('Node search focus failed',err);}
  }

  input.addEventListener('input',render);
  input.addEventListener('focus',()=>{if(normalize(input.value))render();});
  input.addEventListener('keydown',ev=>{
    if(ev.key==='ArrowDown'){ev.preventDefault();if(results.hidden)render();setActive(activeIndex+1);}
    else if(ev.key==='ArrowUp'){ev.preventDefault();if(results.hidden)render();setActive(activeIndex<0?Math.max(0,current.length-1):activeIndex-1);}
    else if(ev.key==='Enter'){if(current.length){ev.preventDefault();choose(activeIndex>=0?activeIndex:0);}}
    else if(ev.key==='Escape'){closeResults();input.blur();}
  });
  clear.addEventListener('click',()=>{input.value='';closeResults();input.focus();});
  document.addEventListener('pointerdown',ev=>{if(!root.contains(ev.target))closeResults();});
})();
